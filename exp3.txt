
clc;
clear;

// System parameters
population = 2e6;
blocking = 0.02;

// System data
cells = [394 98 49];
channels = [19 57 100];

// User traffic
calls_per_hour = 2;
call_duration_min = 3;

traffic_per_user = calls_per_hour * (call_duration_min / 60);

// Erlang B function
function B = erlangB(C, A)
    B = 1;
    for n = 1:C
        B = (A * B) / (n + A * B);
    end
endfunction

// Calculate capacity for each system
for i = 1:3

    C = channels(i);

    // Binary search for traffic A
    low = 0;
    high = C * 2;

    for k = 1:100
        A = (low + high) / 2;
        B = erlangB(C, A);

        if B > blocking then
            high = A;
        else
            low = A;
        end
    end

    A = (low + high) / 2;

    // Total traffic
    total_traffic = cells(i) * A;

    // Users supported
    users = total_traffic / traffic_per_user;

    // Market penetration
    market_penetration = (users / population) * 100;

    mprintf("\nSystem %d\n", i);
    mprintf("Cells = %d\n", cells(i));
    mprintf("Channels per cell = %d\n", channels(i));
    mprintf("Traffic per cell = %.3f Erlangs\n", A);
    mprintf("Total traffic = %.3f Erlangs\n", total_traffic);
    mprintf("Users supported = %.0f\n", users);
    mprintf("Market penetration = %.3f %%\n", market_penetration);

end
