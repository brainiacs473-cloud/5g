function A = erlangA(C, GOS)
    lo = 0;
    hi = C + 50;

    for i = 1:30
        mid = (lo + hi)/2;
        E = 1;

        for k = 1:C
            E = (mid*E)/(k + mid*E);
        end

        if E > GOS
            hi = mid;
        else
            lo = mid;
        end
    end

    A = lo;
end

Pop = 2e6;
GOS = 0.02;
Au = 2*(3/60);

n = input('Enter number of systems: ');

for i = 1:n
    cells = input('Enter number of cells: ');
    ch = input('Enter channels per cell: ');

    d = erlangA(ch, GOS);
    users = (d/Au)*cells;

    fprintf('System %d -> Erlang cap/cell=%.2f | Users=%.0f | Penetration=%.2f%%\n', ...
        i, d, users, users/Pop*100);
end
